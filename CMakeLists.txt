cmake_minimum_required(VERSION 3.20)
project(NeuroGenNative VERSION 0.6.0 LANGUAGES CXX)
option(NEUROGEN_VULKAN "Build Vulkan compute backend" OFF)
option(NEUROGEN_CUDA "Build CUDA compute backend" OFF)
add_library(neurogen_native SHARED src/native/NeuralEngine.cpp src/native/CPUBackend.cpp)
target_include_directories(neurogen_native PUBLIC include PRIVATE src/native)
target_compile_features(neurogen_native PUBLIC cxx_std_17)
if(NEUROGEN_VULKAN)
  find_package(Vulkan REQUIRED)
  find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/lif_spv.h
    COMMAND ${GLSLANG_VALIDATOR} -V ${CMAKE_CURRENT_SOURCE_DIR}/shaders/lif.comp -o ${CMAKE_CURRENT_BINARY_DIR}/lif.spv
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/src/native/embed_shader.py ${CMAKE_CURRENT_BINARY_DIR}/lif.spv ${CMAKE_CURRENT_BINARY_DIR}/lif_spv.h
    DEPENDS shaders/lif.comp src/native/embed_shader.py VERBATIM)
  target_sources(neurogen_native PRIVATE src/native/VulkanBackend.cpp ${CMAKE_CURRENT_BINARY_DIR}/lif_spv.h)
  target_include_directories(neurogen_native PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(neurogen_native PRIVATE Vulkan::Vulkan)
  target_compile_definitions(neurogen_native PRIVATE NEUROGEN_VULKAN)
endif()
if(NEUROGEN_CUDA)
  enable_language(CUDA)
  target_sources(neurogen_native PRIVATE src/native/CUDABackend.cu)
  target_compile_definitions(neurogen_native PRIVATE NEUROGEN_CUDA)
  set_target_properties(neurogen_native PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(native_smoke tests/native/smoke.cpp)
  target_link_libraries(native_smoke PRIVATE neurogen_native)
  add_test(NAME native_cpu COMMAND native_smoke cpu)
  if(NEUROGEN_VULKAN)
    add_test(NAME native_vulkan COMMAND native_smoke vulkan)
  endif()
  if(NEUROGEN_CUDA)
    add_test(NAME native_cuda COMMAND native_smoke cuda)
  endif()
endif()
install(TARGETS neurogen_native LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/engine DESTINATION include FILES_MATCHING PATTERN "ALIFNeuron.h" PATTERN "NeuralEngine.h" PATTERN "CorticalColumnV2.h")
